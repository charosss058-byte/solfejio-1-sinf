const GAME_NAME = "solfejio-1-sinf"; export default { async fetch(request, env) { // CORS const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; // OPTIONS if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: corsHeaders }); } // GET — server tekshiruvi if (request.method === "GET") { return new Response( JSON.stringify({ ok: true, message: "Solfejio license server ishlayapti" }), { headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // Faqat POST if (request.method !== "POST") { return new Response( JSON.stringify({ ok: false, error: "Method not allowed" }), { status: 405, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } try { // Request body const body = await request.json(); const initData = body?.initData; if (!initData) { return new Response( JSON.stringify({ ok: false, access: false, error: "initData yuborilmadi" }), { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // BOT TOKEN kerak const botToken = env.BOT_TOKEN; if (!botToken) { console.error("BOT_TOKEN secret topilmadi"); return new Response( JSON.stringify({ ok: false, access: false, error: "Server konfiguratsiyasi tugallanmagan" }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // Telegram initData ni tekshirish const params = new URLSearchParams(initData); const receivedHash = params.get("hash"); if (!receivedHash) { return new Response( JSON.stringify({ ok: false, access: false, error: "Telegram hash topilmadi" }), { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // hash ni data-check-string ichidan olib tashlaymiz params.delete("hash"); // Telegram qoidasi: // barcha fieldlarni alphabetic tartibda joylaymiz const dataCheckString = [...params.entries()] .sort(([a], [b]) => a.localeCompare(b)) .map(([key, value]) => `${key}=${value}`) .join("\n"); // Secret key const secretKey = await crypto.subtle.importKey( "raw", new TextEncoder().encode("WebAppData"), { name: "HMAC", hash: "SHA-256" }, false, ["sign"] ); const secretKeyBytes = await crypto.subtle.sign( "HMAC", secretKey, new TextEncoder().encode(botToken) ); // Data hash const dataKey = await crypto.subtle.importKey( "raw", secretKeyBytes, { name: "HMAC", hash: "SHA-256" }, false, ["sign"] ); const calculatedHashBytes = await crypto.subtle.sign( "HMAC", dataKey, new TextEncoder().encode(dataCheckString) ); // ArrayBuffer -> hex const calculatedHash = [...new Uint8Array(calculatedHashBytes)] .map(b => b.toString(16).padStart(2, "0")) .join(""); // Hash tekshirish if (calculatedHash !== receivedHash) { console.warn("Telegram hash noto'g'ri"); return new Response( JSON.stringify({ ok: false, access: false, error: "Telegram ma'lumotlari tasdiqlanmadi" }), { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // User ma'lumotini olish const userString = params.get("user"); if (!userString) { return new Response( JSON.stringify({ ok: false, access: false, error: "Telegram user topilmadi" }), { status: 401, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } const telegramUser = JSON.parse(userString); const telegramId = String(telegramUser.id); console.log("Telegram User ID:", telegramId); // D1 dan litsenziyani tekshirish const license = await env.DB .prepare(` SELECT id, telegram_id, game, access, created_at FROM users WHERE telegram_id = ? AND game = ? LIMIT 1 `) .bind(telegramId, GAME_NAME) .first(); // Foydalanuvchi topilmadi if (!license) { return new Response( JSON.stringify({ ok: true, access: false, telegram_id: telegramId, message: "Bu Telegram akkaunt uchun litsenziya topilmadi" }), { headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } // access tekshirish const hasAccess = Number(license.access) === 1; return new Response( JSON.stringify({ ok: true, access: hasAccess, telegram_id: telegramId, game: license.game, message: hasAccess ? "Kirishga ruxsat berildi" : "Kirishga ruxsat berilmagan" }), { headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } catch (error) { console.error("Worker error:", error); return new Response( JSON.stringify({ ok: false, access: false, error: "Server xatosi" }), { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } } ); } } };
